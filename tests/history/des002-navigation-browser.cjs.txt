'use strict';
const {chromium,webkit}=require('playwright'),http=require('node:http'),fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict');
const root=path.resolve('dist'),results=[];
const server=http.createServer((req,res)=>{let p=path.resolve(root,'.'+new URL(req.url,'http://localhost').pathname);if(!p.startsWith(root+path.sep)&&p!==root)return res.writeHead(403).end();if(fs.existsSync(p)&&fs.statSync(p).isDirectory())p=path.join(p,'index.html');if(!fs.existsSync(p))return res.writeHead(404).end();res.setHeader('Content-Type',({'.html':'text/html','.css':'text/css','.js':'text/javascript','.json':'application/json','.svg':'image/svg+xml'})[path.extname(p)]||'application/octet-stream');fs.createReadStream(p).pipe(res);});
(async()=>{await new Promise(r=>server.listen(0,'127.0.0.1',r));const base=`http://127.0.0.1:${server.address().port}`;fs.mkdirSync('test-results/des002',{recursive:true});try{for(const [name,type]of Object.entries({chromium,webkit})){const browser=await type.launch();try{for(const width of [320,375,768,1024,1440])for(const locale of ['','en/']){const page=await browser.newPage({viewport:{width,height:900},hasTouch:true});await page.goto(base+'/'+locale+'?lang='+(locale?'en':'nb'));await page.evaluate(()=>document.fonts.ready);assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));for(const id of ['nav-learn','nav-play']){const trigger=page.locator('#'+id);await trigger.focus();await page.keyboard.press('ArrowDown');assert.equal(await trigger.getAttribute('aria-expanded'),'true');const panel=page.locator('#'+await trigger.getAttribute('aria-controls'));assert(await panel.isVisible());assert(await panel.evaluate(e=>e.contains(document.activeElement)));await page.keyboard.press('Escape');assert.equal(await trigger.getAttribute('aria-expanded'),'false');await trigger.tap();assert.equal(await trigger.getAttribute('aria-expanded'),'true');await panel.locator('a').first().click();assert(page.url().includes('.html'));await page.goto(base+'/'+locale+'?lang='+(locale?'en':'nb'));}
await page.locator('[data-language="'+(locale?'nb':'en')+'"]').click();await page.waitForURL(u=>locale?!u.pathname.startsWith('/en/'):u.pathname.startsWith('/en/'));assert.equal(await page.locator('html').getAttribute('lang'),locale?'nb':'en');await page.goto(base+'/'+locale+'?lang='+(locale?'en':'nb'));assert(await page.locator('.language-picker span').evaluateAll(es=>es.every(e=>getComputedStyle(e).display==='none')));await page.locator('[data-global-search-open]').click();assert(await page.locator('dialog[open]').isVisible());await page.keyboard.press('Escape');await page.locator('.theme-picker summary').click();assert(await page.locator('.theme-options').isVisible());await page.keyboard.press('Escape');await page.goto(base+'/'+locale+'?lang='+(locale?'en':'nb'));await page.evaluate(()=>document.fonts.ready);await page.mouse.move(1,890);await page.screenshot({animations:'disabled',path:`test-results/des002/${name}-${width}-${locale?'en':'nb'}.png`});results.push({browser:name,width,locale:locale?'en':'nb',status:'passed'});await page.close();}}finally{await browser.close();}}}finally{server.close();fs.writeFileSync('test-results/des002/navigation.json',JSON.stringify(results,null,2)+'\n');}console.log(`${results.length} DES-002 navigation/language/mobile checks passed.`);})().catch(e=>{console.error(e);process.exitCode=1;});
